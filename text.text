container.addEventListener('click', e => {
    if(e.target.classList.contains('seat') && !e.target.classList.contains('occupied')){
        e.target.classList.toggle('selected');
        updateSelected()
    }//         ^
});//           |

// ເຮົາຈະນຳເອົາ function ]ທີ່ຢູ່ດ້ານລູ່ມນີ້ສົ່ງຂຶ້ນໄປເຮັດວຽກໃນ function ດ້ານເທິງ
// function ນີ້ແມ່ນຈະຊ່ວຍໃນການນັບຈຳນວນ ແລະ ລາຄາລວມຕາມລາຄາໜັງ ແລະ ຈຳນວນບ່ອນນັ່ງທີ່ເຮົາຄິກຈອງ
function updateSelected() {
    const selectedSeats = document.querySelectorAll('.row .seat.selected');
    const countSeats = selectedSeats.length;

    const seatsIndex = [...selectedSeats].map(seat=>[...seats].indexOf(seat)); //<--------------------------------| ສ່ວນນີ້ແມ່ນໃຊ້ເກັບຂໍ້ມູນບ່ອນນັ່ງລົງໃນ localStarage
    localStorage.setItem("selectedSeats", JSON.stringify(seatsIndex));         //                                 |
                                                                               //                                 |
    count.innerText = countSeats;                                              //                                 |
    total.innerText = countSeats * price;// ----------------|                                                     |
}// function ຈະນຳໃປສະແດງທີ່ <p class="text">                  |                                                     |
//                                                          |                                                     |
//                                                          |                                                     |
//ສ່ວນນີ້ແມ່ນຈະຊ່ວຍໃຫ້ລາຄາປ່ຽນໄປຕາມໜັງທີ່ເຮົາເລືອກ                    ເຮົາຈະນຳເອົາ "function updateSelected()" ມາໃຊ້ຮ່ວມ        |
movieSelect.addEventListener('change', e => {//             |                                                     |
    price = +e.target.value;                 //             |                                                     |
    setMovieData(e.target.selectedIndex, e.target.value);// |                                                     |
    updateSelected(); //  <---------------------------------|                                                     |
}); //     ^                                                                                                      |
    //     |                                                                                                      |
function setMovieData(movieIndex, moviePrice) {                                //                                 |
    localStorage.setItem("movieIndex", movieIndex);                            //                                 |
    localStorage.setItem("moviePrice", moviePrice);//<------------------------------------------------------------| ສ່ວນນີ້ແມ່ນໃຊ້ເກັບຂໍ້ມູນໜັງ ແລະ ລາຄາລົງໃນ localStarage
}

function showDataToUI() {
    const selectedSeats = JSON.parse(localStorage.getItem("selectedSeats"));
    const selectMovieIndex = localStorage.getItem("movieIndex"); // ຕຳແໜ່ງໜັງທີ່ເກັບໄວ້ໃນ localStorage

    seats.forEach((seat,index)=> {
        if(selectedSeats.indexOf(index)>-1) {
            seat.classList.add('selected');
        }
    });
    if(selectMovieIndex !=null) {
        movieSelect.selectedIndex = selectMovieIndex;
    }
}

showDataToUI();
